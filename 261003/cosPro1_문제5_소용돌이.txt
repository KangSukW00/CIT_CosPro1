def solution(n):
	# 빈 네모칸 생성 n x n
	grid = [[0] * n for _ in range(n)]

	# 우,하,좌,상 도구
	dr = [0,1,0,-1] #세로
	dc = [1,0,-1,0]	#가로
	
	r, c, d = 0,0,0

	# 1부터 n^2까지 채우기
	for num in range(1, n*n+1):
		grid[r][c] = num
		
		# 다음 인덱스번호를 정하는 변수
		nr = r + dr[d]
		nc = c + dc[d]
	
	if not (0 <= nr < n and 0 <= nc < n) or grid[nr][nc] != 0:
		d = (d+1)%4
		nr = r + dr[d]
		nc = c + dc[d]
	r, c = nr, nc

	# 대각선 합
	answer = 0
	for i in range(n):
		anser += grid[i][i]

	return answer
